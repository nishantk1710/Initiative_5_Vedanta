# Layout detection (PP-StructureV3) + text OCR (PaddleOCR), scanned path.
# paddlepaddle is pinned to 3.1.0: 3.2+ hits a Windows-CPU oneDNN/PIR executor
# bug (NotImplementedError: ConvertPirAttribute2RuntimeAttribute ...) inside
# the RT-DETR-based layout detector; 3.0.0 avoids that but then fails inside
# PaddleOCR's own text-detection model (ValueError: Type of attribute:
# strides is not right) because its exported op defs are too old. 3.1.0 is
# the version verified to work for both PP-StructureV3 and PaddleOCR here.
paddlepaddle==3.1.0
paddleocr==3.7.0
# paddleocr's PP-StructureV3 needs this extra; version must match paddleocr's
# resolved paddlex dependency (`pip show paddlex`).
paddlex[ocr]==3.7.2

# Handwriting OCR path.
pytesseract
opencv-python

# PaddleOCR-VL-0.9B does NOT run in this environment or this venv — it needs
# paddlepaddle>=3.2.0 (its safetensors loader only handles bfloat16 weights
# natively from that version; below it, loading falls back through numpy,
# which has no bfloat16 type, and raises "data type 'bfloat16' not
# understood"). This service is pinned to paddlepaddle==3.1.0 for PP-
# StructureV3's layout detector, which hits a DIFFERENT bug on
# paddlepaddle>=3.2 on Windows CPU. Both constraints are real, so
# PaddleOCR-VL runs as a separate microservice in its own venv — see
# python_vl_service/requirements.txt and its module docstring.

# Tests (python/tests/). Fast suite runs in ~26s; `-m slow` runs the real
# OCR pipeline and takes ~20 min. See python/tests/README.md.
pytest

#   pip install -r requirements.txt --break-system-packages

# pytesseract is only a wrapper — it shells out to the real Tesseract binary,

#                   (or the installer at https://github.com/UB-Mannheim/tesseract/wiki)
#   Debian/Ubuntu:  sudo apt install tesseract-ocr
#   macOS:          brew install tesseract
